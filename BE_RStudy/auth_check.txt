const register = async (req, res, next) => {

    const account = await authService.register(req.body);
    return res.status(201).json(formatResponse(true, account, "Đăng ký thành công"));

  try {
    const result = await authService.login(req.body.email, req.body.password);
    return res.status(200).json(formatResponse(true, result, "Đăng nhập thành công"));
  } catch (err) {
    next(err);
  }

  try {
    const account = await authService.getMe(req.user.id);
    return res.status(200).json(formatResponse(true, account));
  } catch (err) {
    next(err);
  }
};

===== src/services/authService.js =====
const authRepository = require("../repositories/authRepository");
const { hashPassword, comparePassword } = require("../utils/hash");
const { signAccessToken } = require("../utils/jwt");
const AppError = require("../utils/AppError");

const register = async (payload) => {
  const existing = await authRepository.findByEmail(payload.email);
  if (existing) throw new AppError("Email đã được sử dụng", 409);

  const passwordHash = await hashPassword(payload.password);
  const data = { ...payload, passwordHash };

  return payload.role === "TEACHER" ? authRepository.createTeacherAccount(data) : authRepository.createStudentAccount(data);
};

const login = async (email, password) => {
  const account = await authRepository.findByEmail(email);
  if (!account) throw new AppError("Email hoặc mật khẩu không đúng", 401);

  const match = await comparePassword(password, account.passwordHash);
  if (!match) throw new AppError("Email hoặc mật khẩu không đúng", 401);

  if (account.status === "BI_KHOA") throw new AppError("Tài khoản đã bị khóa", 403);

  // CHO_XAC_MINH vẫn được đăng nhập (OTP xác minh mail chưa làm — STT 14)
  const accessToken = signAccessToken({ sub: account.id, role: account.role });

  const { passwordHash, ...safeAccount } = account;
  return { accessToken, account: safeAccount };
};

const getMe = async (accountId) => {
  const account = await authRepository.findByIdSafe(accountId);
  if (!account) throw new AppError("Không tìm thấy tài khoản", 404);
  return account;
};

module.exports = { register, login, getMe };

===== src/middlewares/authMiddleware.js =====
const { verifyAccessToken } = require("../utils/jwt");
const AppError = require("../utils/AppError");

// authenticate: xác định req.user = { id, role }
// Khi DEV_AUTH_BYPASS=true và KHÔNG có header Authorization, tạo user giả
// qua header x-dev-role (STUDENT/TEACHER/ADMIN), mặc định STUDENT — để các
// module khác test được API cần đăng nhập trước khi Auth thật xong.
const authenticate = (req, res, next) => {
  const header = req.headers.authorization;

  if (!header) {
    if (process.env.DEV_AUTH_BYPASS === "true") {
      req.user = {
        id: req.headers["x-dev-user-id"] || "dev-user-id",
        role: req.headers["x-dev-role"] || "STUDENT",
      };
      return next();
    }
    return next(new AppError("Thiếu token xác thực", 401));
  }

  const token = header.replace("Bearer ", "");
  try {
    const payload = verifyAccessToken(token);
    req.user = { id: payload.sub, role: payload.role };
    next();
  } catch (err) {
    next(new AppError("Token không hợp lệ hoặc hết hạn", 401));
  }
};

// authorize('TEACHER', 'ADMIN') — chặn nếu role không nằm trong danh sách
const authorize =
  (...roles) =>
  (req, res, next) => {
    if (!req.user) return next(new AppError("Chưa xác thực", 401));
    if (!roles.includes(req.user.role)) {
      return next(new AppError("Không đủ quyền truy cập", 403));
    }
    next();
  };

module.exports = { authenticate, authorize };

const jwt = require("jsonwebtoken");

const signAccessToken = (payload) =>
  jwt.sign(payload, process.env.JWT_ACCESS_SECRET, {
    expiresIn: process.env.JWT_ACCESS_EXPIRES || "15m",
  });

const signRefreshToken = (payload) =>
  jwt.sign(payload, process.env.JWT_REFRESH_SECRET, {
    expiresIn: process.env.JWT_REFRESH_EXPIRES || "7d",
  });

const verifyAccessToken = (token) => jwt.verify(token, process.env.JWT_ACCESS_SECRET);
const verifyRefreshToken = (token) => jwt.verify(token, process.env.JWT_REFRESH_SECRET);

module.exports = { signAccessToken, signRefreshToken, verifyAccessToken, verifyRefreshToken };
