===== src/api/axiosClient.js =====
import axios from 'axios';
import { tokenStorage } from '../utils/tokenStorage';

export const AUTH_LOGOUT_EVENT = 'auth:logout';

const baseURL = import.meta.env.VITE_API_URL;
const NO_REFRESH_URLS = ['/auth/login', '/auth/refresh'];

const axiosClient = axios.create({
  baseURL,
  headers: { 'Content-Type': 'application/json' },
});

let refreshPromise = null;

const refreshTokens = () => {
  if (!refreshPromise) {
    refreshPromise = axios
      .post(`${baseURL}/auth/refresh`, { refreshToken: tokenStorage.getRefresh() })
      .then((res) => {
        const { accessToken, refreshToken } = res.data.data;
        tokenStorage.set(accessToken, refreshToken);
        return accessToken;
      })
      .finally(() => {
        refreshPromise = null;
      });
  }
  return refreshPromise;
};

axiosClient.interceptors.request.use((config) => {
  const accessToken = tokenStorage.getAccess();
  if (accessToken) {
    config.headers.Authorization = `Bearer ${accessToken}`;
  }
  return config;
});

axiosClient.interceptors.response.use(
  (response) => response.data,
  async (error) => {
    const original = error.config;
    const isNoRefreshUrl = NO_REFRESH_URLS.some((url) => original?.url?.startsWith(url));

    if (
      error.response?.status === 401 &&
      original &&
      !original._retry &&
      !isNoRefreshUrl &&
      tokenStorage.getRefresh()
    ) {
      original._retry = true;
      try {
        const accessToken = await refreshTokens();
        original.headers.Authorization = `Bearer ${accessToken}`;
        return axiosClient(original);
      } catch {
        tokenStorage.clear();
        window.dispatchEvent(new Event(AUTH_LOGOUT_EVENT));
      }
    }

    error.message = error.response?.data?.message || 'Không thể kết nối máy chủ';
    return Promise.reject(error);
  },
);

export default axiosClient;

===== src/context/AuthContext.jsx =====
import { useEffect, useState } from 'react';
import { authApi } from '../api/authApi';
import { AUTH_LOGOUT_EVENT } from '../api/axiosClient';
import { tokenStorage } from '../utils/tokenStorage';
import { AuthContext } from './authContextValue';

export function AuthProvider({ children }) {
  const [account, setAccount] = useState(null);
  const [loading, setLoading] = useState(() => Boolean(tokenStorage.getRefresh()));

  useEffect(() => {
    if (!tokenStorage.getRefresh()) return undefined;
    let active = true;
    authApi
      .getMe()
      .then((res) => {
        if (active) setAccount(res.data);
      })
      .catch(() => {
        tokenStorage.clear();
      })
      .finally(() => {
        if (active) setLoading(false);
      });
    return () => {
      active = false;
    };
  }, []);

  useEffect(() => {
    const handleLogout = () => setAccount(null);
    window.addEventListener(AUTH_LOGOUT_EVENT, handleLogout);
    return () => window.removeEventListener(AUTH_LOGOUT_EVENT, handleLogout);
  }, []);

  const login = async (credentials) => {
    const loginRes = await authApi.login(credentials);
    tokenStorage.set(loginRes.data.accessToken, loginRes.data.refreshToken);
    try {
      const meRes = await authApi.getMe();
      setAccount(meRes.data);
      return meRes.data;
    } catch (error) {
      tokenStorage.clear();
      throw error;
    }
  };

  const logout = async () => {
    const refreshToken = tokenStorage.getRefresh();
    if (refreshToken) await authApi.logout(refreshToken).catch(() => null);
    tokenStorage.clear();
    setAccount(null);
  };

  const value = { account, loading, login, logout, updateAccount: setAccount };

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

===== src/utils/tokenStorage.js =====
const ACCESS_KEY = 'accessToken';
const REFRESH_KEY = 'refreshToken';

export const tokenStorage = {
  getAccess: () => localStorage.getItem(ACCESS_KEY),
  getRefresh: () => localStorage.getItem(REFRESH_KEY),
  set: (accessToken, refreshToken) => {
    localStorage.setItem(ACCESS_KEY, accessToken);
    localStorage.setItem(REFRESH_KEY, refreshToken);
  },
  clear: () => {
    localStorage.removeItem(ACCESS_KEY);
    localStorage.removeItem(REFRESH_KEY);
  },
};

===== src/components/ProtectedRoute.jsx =====
import { Loader2 } from 'lucide-react';
import { Navigate, Outlet, useLocation } from 'react-router-dom';
import { ROLE_HOME, ROUTES } from '../constants/routes';
import { useAuth } from '../hooks/useAuth';

export default function ProtectedRoute({ allowedRoles }) {
  const { account, loading } = useAuth();
  const location = useLocation();

  if (loading) {
    return (
      <div className="flex min-h-screen items-center justify-center">
        <Loader2 size={24} className="animate-spin" />
      </div>
    );
  }

  if (!account) {
    return <Navigate to={ROUTES.LOGIN} replace state={{ from: location }} />;
  }

  if (allowedRoles && !allowedRoles.includes(account.role)) {
    return <Navigate to={ROLE_HOME[account.role] ?? ROUTES.HOME} replace />;
  }

  return <Outlet />;
}
